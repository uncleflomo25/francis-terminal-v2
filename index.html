import numpy as np import pandas as pd # ============================= # MOCK DATA INPUT (replace later with API) # ============================= def get_data(): data = { "USDT.D": np.random.normal(7, 0.1, 100), "BTC.D": np.random.normal(60, 0.3, 100), "ETHBTC": np.random.normal(0.03, 0.001, 100), "TOTAL": np.random.normal(2.5, 0.05, 100), } return pd.DataFrame(data) # ============================= # INDICATORS # ============================= def ema(series, period=50): return series.ewm(span=period, adjust=False).mean() def roc(series, period=10): return series.pct_change(period) # ============================= # SIGNAL FUNCTION # ============================= def calc_signal(series): trend = np.where(series > ema(series), 1, -1) momentum = roc(series).fillna(0) return trend * momentum # ============================= # REGIME DETECTION # ============================= def detect_regime(df): btc_up = ema(df["BTC.D"], 50).iloc[-1] > ema(df["BTC.D"], 100).iloc[-1] usdt_up = ema(df["USDT.D"], 50).iloc[-1] > ema(df["USDT.D"], 100).iloc[-1] total_up = ema(df["TOTAL"], 50).iloc[-1] > ema(df["TOTAL"], 100).iloc[-1] if usdt_up: return "RISK_OFF" elif btc_up: return "BTC_DOM" elif total_up: return "ALT" else: return "CHOP" # ============================= # ADAPTIVE WEIGHTS # ============================= def get_weights(regime): if regime == "BTC_DOM": return {"USDT":0.15,"BTC":0.4,"ETH":0.2,"TOTAL":0.25} elif regime == "ALT": return {"USDT":0.15,"BTC":0.15,"ETH":0.4,"TOTAL":0.3} elif regime == "RISK_OFF": return {"USDT":0.5,"BTC":0.2,"ETH":0.1,"TOTAL":0.2} else: return {"USDT":0.25,"BTC":0.25,"ETH":0.25,"TOTAL":0.25} # ============================= # SCORE ENGINE # ============================= def compute_score(df, weights): usdt_sig = calc_signal(df["USDT.D"]).iloc[-1] btc_sig = calc_signal(df["BTC.D"]).iloc[-1] eth_sig = calc_signal(df["ETHBTC"]).iloc[-1] tot_sig = calc_signal(df["TOTAL"]).iloc[-1] raw = ( (-usdt_sig * weights["USDT"]) + ( eth_sig * weights["ETH"]) + ( btc_sig * weights["BTC"]) + ( tot_sig * weights["TOTAL"]) ) score = max(0, min(100, 50 + raw * 50)) return score, usdt_sig, btc_sig, eth_sig, tot_sig # ============================= # CONFIDENCE # ============================= def compute_confidence(signals): aligned = sum(np.sign(s) == np.sign(signals[-1]) for s in signals) strength = sum(abs(s) for s in signals) return min(100, aligned * 20 + strength * 10) # ============================= # FINAL SIGNAL # ============================= def classify(score): if score > 75: return "STRONG LONG" elif score > 60: return "LONG" elif score < 25: return "STRONG SHORT" elif score < 40: return "SHORT" else: return "NO TRADE" # ============================= # RUN ENGINE # ============================= def run_engine(): df = get_data() regime = detect_regime(df) weights = get_weights(regime) score, usdt, btc, eth, total = compute_score(df, weights) confidence = compute_confidence([usdt, btc, eth, total]) signal = classify(score) print("\n⚡ V10 ADAPTIVE ENGINE") print("======================") print(f"Regime: {regime}") print(f"Score: {score:.2f}") print(f"Confidence: {confidence:.2f}") print(f"Signal: {signal}") # ============================= # START # ============================= if __name__ == "__main__": run_engine()
