const express = require("express"); const axios = require("axios"); const cors = require("cors"); require("dotenv").config(); const app = express(); app.use(cors()); app.use(express.json()); const BASE_URL = "https://testnet.binance.vision"; const API_KEY = process.env.BINANCE_API_KEY; // ===================== // FRONTEND (SINGLE PAGE) // ===================== const frontend = ` FRANCIS TERMINAL LIVE

BTCUSDT

Loading...

`; // ===================== // SERVE FRONTEND // ===================== app.get("/", (req,res)=>{ res.send(frontend); }); // ===================== // LIVE PRICE (BINANCE) // ===================== app.get("/price/:symbol", async (req,res)=>{ try{ const symbol = req.params.symbol; const r = await axios.get( `${BASE_URL}/api/v3/ticker/price?symbol=${symbol}` ); res.json(r.data); }catch(e){ res.json({error:"price failed"}); } }); // ===================== // PLACE ORDER (TESTNET) // ===================== app.post("/order", async (req,res)=>{ try{ const {symbol,side,quantity} = req.body; const r = await axios.post( `${BASE_URL}/api/v3/order`, null, { params:{ symbol, side, type:"MARKET", quantity, timestamp:Date.now() }, headers:{ "X-MBX-APIKEY": API_KEY } } ); res.json(r.data); }catch(e){ res.json({error:"order failed",msg:e.message}); } }); // ===================== app.listen(3000,()=>{ console.log("TRADING TERMINAL RUNNING http://localhost:3000"); });
