//@version=5 indicator("⚡ FRANCIS TERMINAL V10 — ADAPTIVE WEIGHTED ENGINE", overlay=true, max_labels_count=500) //============================= // SYMBOLS //============================= usdtSym = input.symbol("CRYPTOCAP:USDT.D", "USDT.D") btcDSym = input.symbol("CRYPTOCAP:BTC.D", "BTC.D") ethbtcSym= input.symbol("BINANCE:ETHBTC", "ETHBTC") totalSym = input.symbol("CRYPTOCAP:TOTAL", "TOTAL") tfHTF = input.timeframe("D", "HTF (Daily)") //============================= // DATA //============================= usdt = request.security(usdtSym, timeframe.period, close) btcD = request.security(btcDSym, timeframe.period, close) ethbtc= request.security(ethbtcSym,timeframe.period, close) total = request.security(totalSym, timeframe.period, close) vol_usdt = request.security(usdtSym, timeframe.period, volume) vol_btcd = request.security(btcDSym, timeframe.period, volume) vol_ethbtc= request.security(ethbtcSym,timeframe.period, volume) vol_total = request.security(totalSym, timeframe.period, volume) // HTF usdt_htf = request.security(usdtSym, tfHTF, close) btcD_htf = request.security(btcDSym, tfHTF, close) ethbtc_htf= request.security(ethbtcSym,tfHTF, close) total_htf = request.security(totalSym, tfHTF, close) //============================= // TREND + MOMENTUM //============================= f_signal(src) => ema = ta.ema(src, 50) trend = src > ema ? 1 : -1 mom = ta.roc(src, 10) vol = volume > ta.sma(volume, 20) ? 1 : 0.7 trend * (mom / 5) * vol usdt_sig = f_signal(usdt) btcD_sig = f_signal(btcD) ethbtc_sig = f_signal(ethbtc) total_sig = f_signal(total) //============================= // REGIME DETECTION //============================= btcDomTrend = ta.ema(btcD, 50) > ta.ema(btcD, 100) usdtTrend = ta.ema(usdt, 50) > ta.ema(usdt, 100) totalTrend = ta.ema(total, 50) > ta.ema(total, 100) regime = usdtTrend ? "RISK_OFF" : btcDomTrend ? "BTC_DOM" : totalTrend ? "ALT" : "CHOP" //============================= // ADAPTIVE WEIGHTS //============================= float w_usdt = 0.0 float w_btcd = 0.0 float w_eth = 0.0 float w_tot = 0.0 if regime == "BTC_DOM" w_btcd := 0.4 w_tot := 0.25 w_eth := 0.2 w_usdt := 0.15 else if regime == "ALT" w_eth := 0.4 w_tot := 0.3 w_btcd := 0.15 w_usdt := 0.15 else if regime == "RISK_OFF" w_usdt := 0.5 w_btcd := 0.2 w_tot := 0.2 w_eth := 0.1 else w_usdt := 0.25 w_btcd := 0.25 w_eth := 0.25 w_tot := 0.25 //============================= // SCORE ENGINE //============================= rawScore = (usdt_sig * -w_usdt) + (ethbtc_sig * w_eth ) + (btcD_sig * w_btcd) + (total_sig * w_tot ) score = math.max(0, math.min(100, 50 + rawScore * 25)) //============================= // HTF FILTER //============================= htfTrend = total_htf > ta.ema(total_htf, 50) ? 1 : -1 score := htfTrend > 0 ? score : score * 0.7 //============================= // CONFIDENCE //============================= alignment = (math.sign(usdt_sig) == math.sign(total_sig) ? 1 : 0) + (math.sign(ethbtc_sig) == math.sign(total_sig) ? 1 : 0) + (math.sign(btcD_sig) == math.sign(total_sig) ? 1 : 0) confidence = alignment * 25 + math.abs(rawScore) * 20 confidence := math.min(confidence, 100) //============================= // FINAL SIGNAL //============================= string signal = score > 75 ? "STRONG LONG" : score > 60 ? "LONG" : score < 25 ? "STRONG SHORT" : score < 40 ? "SHORT" : "NO TRADE" //============================= // COLORS //============================= bgcolor(score > 70 ? color.new(color.green, 90) : score < 30 ? color.new(color.red, 90) : na) //============================= // DASHBOARD //============================= var table dash = table.new(position.top_right, 2, 6, bgcolor=color.black) if barstate.islast table.cell(dash, 0, 0, "⚡ V10 ENGINE", text_color=color.yellow) table.cell(dash, 1, 0, "") table.cell(dash, 0, 1, "Regime") table.cell(dash, 1, 1, regime, text_color=color.cyan) table.cell(dash, 0, 2, "Score") table.cell(dash, 1, 2, str.tostring(score, "#.0"), text_color=color.green) table.cell(dash, 0, 3, "Confidence") table.cell(dash, 1, 3, str.tostring(confidence, "#.0"), text_color=color.orange) table.cell(dash, 0, 4, "Signal") table.cell(dash, 1, 4, signal, text_color=color.white) table.cell(dash, 0, 5, "Vol TOTAL") table.cell(dash, 1, 5, str.tostring(vol_total, "#.##")) //============================= // SIGNAL MARKERS //============================= plotshape(score > 75, title="BUY", location=location.belowbar, color=color.green, style=shape.labelup, text="BUY") plotshape(score < 25, title="SELL", location=location.abovebar, color=color.red, style=shape.labeldown, text="SELL")
